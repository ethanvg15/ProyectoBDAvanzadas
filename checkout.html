@eventos_bp.route("/reservar/", methods=["POST"]) def reservar_entrada(id_evento): """ Fase 1 (Estado Activo / Bloqueo Temporal en Redis con TTL): Verifica el inventario real descontando reservas activas en RAM y bloquea el cupo. """ db = get_mongo_db() redis_client = get_redis_client() cantidad_solicitada = int(request.form.get("cantidad", 1)) if "cliente_id" not in session: session["cliente_id"] = str(uuid.uuid4())[:8] cliente_id = session["cliente_id"] # 1. Validar que la preventa siga activa en Redis if not redis_client.exists(f"preventa_activa:{id_evento}"): flash("La ventana de preventa para este evento ha expirado o no está activa.", "error") return redirect(url_for("eventos.ver_catalogo")) # 2. Consultar inventario persistente en MongoDB evento = db.eventos.find_one({"id_evento": id_evento}) if not evento: flash("El evento seleccionado no existe.", "error") return redirect(url_for("eventos.ver_catalogo")) # 3. Calcular cupos retenidos en Redis por otros usuarios concurrentes llaves_reservas = redis_client.keys(f"reserva:{id_evento}:*") entradas_retenidas = sum(int(redis_client.get(k) or 0) for k in llaves_reservas) disponibles_reales = int(evento.get("entradas_disponibles", 0)) - entradas_retenidas if cantidad_solicitada > disponibles_reales: flash(f"Cupo insuficiente. Solo quedan {disponibles_reales} entradas disponibles.", "error") return redirect(url_for("eventos.ver_catalogo")) # 4. Crear bloqueo temporal en Redis con TTL de 120 segundos tiempo_ttl = 120 llave_usuario = f"reserva:{id_evento}:{cliente_id}" redis_client.setex(llave_usuario, tiempo_ttl, cantidad_solicitada) flash(f"Reserva temporal activada por {tiempo_ttl} segundos.", "success") return redirect(url_for("eventos.checkout_reserva", id_evento=id_evento)) @eventos_bp.route("/checkout/", methods=["GET"]) def checkout_reserva(id_evento): """Muestra la ventana de confirmación mientras el TTL de Redis esté vigente.""" db = get_mongo_db() redis_client = get_redis_client() cliente_id = session.get("cliente_id") llave_usuario = f"reserva:{id_evento}:{cliente_id}" ttl_restante = redis_client.ttl(llave_usuario) cantidad_reservada = redis_client.get(llave_usuario) if ttl_restante <= 0 or not cantidad_reservada: flash("Transacción abortada: Tu reserva temporal expiró y los cupos fueron liberados.", "error") return redirect(url_for("eventos.ver_catalogo")) evento = db.eventos.find_one({"id_evento": id_evento}) return render_template( "eventos/checkout.html", evento=evento, cantidad=int(cantidad_reservada), ttl=ttl_restante ) @eventos_bp.route("/confirmar_compra/", methods=["POST"]) def confirmar_compra(id_evento): """ Fase 2 (Estado Committed / Persistencia en MongoDB): Consolida la venta en disco y libera la llave temporal de Redis. """ db = get_mongo_db() redis_client = get_redis_client() cliente_id = session.get("cliente_id") llave_usuario = f"reserva:{id_evento}:{cliente_id}" cantidad_str = redis_client.get(llave_usuario) if not cantidad_str: flash("Transacción abortada: El tiempo límite expiró antes de confirmar.", "error") return redirect(url_for("eventos.ver_catalogo")) cantidad_comprada = int(cantidad_str) evento = db.eventos.find_one({"id_evento": id_evento}) # 1. Descontar inventario en MongoDB de forma atómica db.eventos.update_one( {"id_evento": id_evento}, {"$inc": {"entradas_disponibles": -cantidad_comprada}} ) # 2. Registrar comprobante en la colección 'ventas' (Durabilidad) db.ventas.insert_one({ "id_venta": str(uuid.uuid4())[:12], "id_evento": id_evento, "nombre_evento": evento.get("nombre"), "cliente_id": cliente_id, "cantidad": cantidad_comprada, "total_pagado": cantidad_comprada * float(evento.get("precio", 0)) }) # 3. Liberar el bloqueo en Redis redis_client.delete(llave_usuario) flash("¡Compra confirmada (Committed) exitosamente!", "success") return redirect(url_for("eventos.ver_catalogo"))
